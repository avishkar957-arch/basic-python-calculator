a = float(input("Enter first number: "))
b = input("Enter a operation (+, -, *, /): ")
c = float(input("Enter second number: "))

if b == "+":
    print(a + c)
elif b == "-":
    print(a - c)
elif b == "*":
    print(a * c)
elif b == "/":
    print(a / c)
else:
    print("Invalid operator")
